"""KGE-Forge sealed profile recovery: fail-closed operator integration scaffold.

Preflight is read-only. Execution is DISABLED until the governed consumption

ambiguous consumption/write outcome.

import argparse
import fcntl
import hashlib
import os
import stat
import sys

SOURCE = '/home/gvasend/app/kge-forge/docs/plans/governance/development-model-adoption-standing-grant-issuance/restricted-permission-profile.toml.txt'
DEST = '/home/gvasend/.codex/kge-forge-class-a.config.toml'
SOURCE_SHA = 'b694333f4c108c8fea455f7e117cda1d003f3c915bb36d78ae7ec466022b47f8'
EMPTY_SHA = hashlib.sha256(b'').hexdigest()
ATTEMPT_KEY = '48f7ff401d5bfec3a300f79f2bab28eece53e0243865994d54fc2ae58c8c1e1f'
SIZE = 1091
EXPECTED_UID = EXPECTED_GID = 1000
EXPECTED_MODE = 0o600

class Stop(Exception):
    pass

def require(ok, msg):
    if not ok:
        raise Stop(msg)

def read_source():
    # Read only; source parent chain still needs host-side trusted qualification.
    with open(SOURCE, 'rb') as f:
        data = f.read(SIZE + 1)
        require(f.read(1) == b'', 'source unexpectedly longer')
    require(len(data) == SIZE, 'source length mismatch')
    require(hashlib.sha256(data).hexdigest() == SOURCE_SHA, 'source hash mismatch')
    return data

def inspect_destination():
    # O_PATH-style parent traversal with no symlinks; no destination write open.
    flags = os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW | os.O_CLOEXEC
    root = os.open('/', flags)
    current = root
    try:
        for component in ['home', 'gvasend', '.codex']:
            nextfd = os.open(component, flags, dir_fd=current)
            if current != root:
                os.close(current)
            current = nextfd
        fd = os.open('kge-forge-class-a.config.toml', os.O_RDONLY | os.O_NOFOLLOW | os.O_CLOEXEC | os.O_NONBLOCK, dir_fd=current)
        try:
            s = os.fstat(fd)
            require(stat.S_ISREG(s.st_mode), 'destination is not regular')
            require(s.st_uid == EXPECTED_UID and s.st_gid == EXPECTED_GID, 'destination owner mismatch')
            require(stat.S_IMODE(s.st_mode) == EXPECTED_MODE, 'destination mode mismatch')
            require(s.st_nlink == 1 and s.st_size == 0, 'destination link/size mismatch')
            require(hashlib.sha256(os.read(fd, 1)).hexdigest() == EMPTY_SHA, 'destination not empty')
            return (s.st_dev, s.st_ino)
        finally:
            os.close(fd)
    finally:
        if current != root:
            os.close(current)
        os.close(root)

def durable_consume_and_independently_verify(attempt_key, device, inode):
    """Integration boundary: the sealed governance custodian's exclusive,
    durable attempt-consumption transaction, including independent reread.

    This MUST be replaced only with an authenticated implementation bound to
    the effective custody supplement and authoritative record domain.
    Do not use a local marker, shell flag, or unverified receipt as a substitute.
    """
    raise Stop('BLOCKED: governed durable-consumption adapter not supplied')

def main():
    p = argparse.ArgumentParser(description=__doc__)
    p.add_argument('--preflight', action='store_true', help='read-only snapshot only')
    p.add_argument('--execute', action='store_true', help='disabled pending governed integration')
    args = p.parse_args()
    require(args.preflight != args.execute, 'choose exactly one mode')
    require(os.geteuid() == EXPECTED_UID and os.getegid() == EXPECTED_GID, 'operator identity mismatch')
    read_source()
    dev, ino = inspect_destination()
    print('PREFLIGHT_SNAPSHOT_PASS device={} inode={} (not launch authorization)'.format(dev, ino))
    if args.execute:
        # No writable open or write occurs without the governed consumption adapter.
        durable_consume_and_independently_verify(ATTEMPT_KEY, dev, ino)
        raise Stop('BLOCKED: complete sealed descriptor-bound write implementation not installed')

if __name__ == '__main__':
    try:
        main()
    except (Stop, OSError) as exc:
        print('STOP: {}'.format(exc), file=sys.stderr)
        sys.exit(2)
