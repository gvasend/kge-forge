def read(p):return json.loads(pathlib.Path(p).read_text())
def digest(x,ascii=True):return hashlib.sha256(json.dumps(x,sort_keys=True,separators=(',',':'),ensure_ascii=ascii).encode()).hexdigest()
def raw(p):return hashlib.sha256(pathlib.Path(p).read_bytes()).hexdigest()
def checkpin(p):assert raw(p['path'])==p['raw_sha256']
subject=read(R/'owner-decision-subject.json');policy=read(B/'engineering-policy-supplement.json');contract=read(R/'standing-development-contract.json');pilot=read(R/'pilot-authorization-subject.json');approval=read(D/'approval-authentication.json')

assert approval['SUBJECT_IDENTITY']==subject['IDENTITY'] and approval['OPTION']=='ADOPT_MODEL_AND_AUTHORIZE_CONDITIONAL_CLASS_A_DEVELOPMENT_PILOT' and approval['OWNER_ROLE']=='KGE-FORGE-PROJECT-OWNER'
checkpin(approval['SUBJECT_RAW_PIN'])
for section,labels in [(subject['CONTENT']['GOVERNANCE_SECTION'],['POLICY']),(subject['CONTENT']['STANDING_EXECUTION_SECTION'],['CONTRACT','PILOT'])]:

 v=read(t['PATH']);assert digest(v['RECORD'],False)==v['IDENTITY']['sha256'];assert v['IDENTITY']==t['IDENTITY'] and v['RECORD']['to_state']==state
 if previous is not None:
  # Whole linked transition bytes are additionally pinned by prior-file audit;
  # exact predecessor equality is reconstructed from the recorded chain field.
  assert previous['sha256'] in json.dumps(v['RECORD'])
 previous=v['IDENTITY']
assert previous==charter['EFFECTIVE_LIFECYCLE_IDENTITY']
events=read('docs/plans/PLANNER_REASONING_LESSONS_REGISTER_1.json')['EVENTS'];prev='0'*64
